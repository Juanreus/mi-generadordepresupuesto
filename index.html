function readInputsToState() { state.config.dolar = parseFloat(document.getElementById("cfg-dolar").value) || 0; state.config.margenPct = parseFloat(document.getElementById("cfg-margen").value) || 0; state.config.redondeo = parseFloat(document.getElementById("cfg-redondeo").value) || 0; state.meta.cliente = document.getElementById("meta-cliente").value; state.meta.domicilio = document.getElementById("meta-domicilio").value; state.meta.contacto = document.getElementById("meta-contacto").value; state.meta.obra = document.getElementById("meta-obra").value; state.meta.fecha = document.getElementById("meta-fecha").value; state.meta.validez = parseFloat(document.getElementById("meta-validez").value) || 0; state.meta.descuentoPct = parseFloat(document.getElementById("meta-descuento").value) || 0; state.meta.ivaPct = parseFloat(document.getElementById("meta-iva").value) || 0; state.condiciones.pago = document.getElementById("cond-pago").value; state.condiciones.plazo = document.getElementById("cond-plazo").value; state.condiciones.garantia = document.getElementById("cond-garantia").value; state.condiciones.noIncluye = document.getElementById("cond-noincluye").value; state.condiciones.obs = document.getElementById("cond-obs").value; } function crearSnapshotPresupuesto() { return { config: { dolar: state.config.dolar, margenPct: state.config.margenPct, redondeo: state.config.redondeo }, meta: JSON.parse(JSON.stringify(state.meta)), condiciones: JSON.parse(JSON.stringify(state.condiciones)), items: JSON.parse(JSON.stringify(state.items)) }; } function guardarEnHistorial(forceUpdate = false) { readInputsToState(); actualizarVisibilidadCondiciones(); const res = calcularResumen(); const snapshot = crearSnapshotPresupuesto(); if (editingHistorialIndex !== null && state.historial[editingHistorialIndex]) { const existing = state.historial[editingHistorialIndex]; state.historial[editingHistorialIndex] = { ...existing, fecha: state.meta.fecha || existing.fecha, cliente: state.meta.cliente || "Sin especificar", obra: state.meta.obra || "Presupuesto general", total: res.totalFinal, snapshot }; renderHistorial(); renderPresupuesto(); saveState(); showToast(`Presupuesto ${existing.numero} actualizado`); return; } const numero = getCurrentBudgetNumber(); state.historial.push({ numero, fecha: state.meta.fecha || new Date().toISOString().slice(0, 10), cliente: state.meta.cliente || "Sin especificar", obra: state.meta.obra || "Presupuesto general", total: res.totalFinal, estado: "Pendiente", obs: "Registrado desde web", snapshot }); editingHistorialIndex = state.historial.length - 1; actualizarBannerEdicion(); renderHistorial(); renderPresupuesto(); saveState(); showToast(`Guardado ${numero} en el historial`); } function editarHistorial(idx) { const reg = state.historial[idx]; if (!reg) return; editingHistorialIndex = idx; if (reg.snapshot) { if (reg.snapshot.config) { state.config.dolar = reg.snapshot.config.dolar ?? state.config.dolar; state.config.margenPct = reg.snapshot.config.margenPct ?? state.config.margenPct; state.config.redondeo = reg.snapshot.config.redondeo ?? state.config.redondeo; } if (reg.snapshot.meta) { state.meta = { ...DEFAULT_DATA.meta, ...JSON.parse(JSON.stringify(reg.snapshot.meta)) }; } if (reg.snapshot.condiciones) { state.condiciones = { ...DEFAULT_DATA.condiciones, ...JSON.parse(JSON.stringify(reg.snapshot.condiciones)) }; } if (Array.isArray(reg.snapshot.items) && reg.snapshot.items.length > 0) { state.items = JSON.parse(JSON.stringify(reg.snapshot.items)); } } else { state.meta = { ...state.meta, cliente: reg.cliente || "", obra: reg.obra || "", fecha: reg.fecha || state.meta.fecha, descuentoPct: 0, ivaPct: 0 }; state.items = [ { id: Date.now(), cant: 1, desc: reg.obra || "Detalle del presupuesto", usd: null, arsDir: Number(reg.total) || 0, margenPct: 0 } ]; } syncInputsFromState(); actualizarBannerEdicion(); renderAll(); switchTab("armador"); showToast(`Editando ${reg.numero}`); }
